skoed has been deployed.

1. Wait for the pod to be ready:

   kubectl wait --for=condition=ready pod \
     -l app.kubernetes.io/instance={{ .Release.Name }} \
     -n {{ .Release.Namespace }} \
     --timeout=60s

2. Access the management API:

{{- if eq .Values.service.type "ClusterIP" }}

   kubectl port-forward svc/{{ include "skoed.fullname" . }} 8080:{{ .Values.service.apiPort }} -n {{ .Release.Namespace }}

   Then open http://localhost:8080 in your browser.
{{- else if eq .Values.service.type "NodePort" }}

   export NODE_IP=$(kubectl get nodes -o jsonpath="{.items[0].status.addresses[0].address}")
   export NODE_PORT=$(kubectl get svc {{ include "skoed.fullname" . }} -n {{ .Release.Namespace }} -o jsonpath="{.spec.ports[?(@.name=='api')].nodePort}")
   echo "http://$NODE_IP:$NODE_PORT"
{{- else if eq .Values.service.type "LoadBalancer" }}

   export LB_IP=$(kubectl get svc {{ include "skoed.fullname" . }} -n {{ .Release.Namespace }} -o jsonpath="{.status.loadBalancer.ingress[0].ip}")
   echo "http://$LB_IP:{{ .Values.service.apiPort }}"
{{- end }}

3. Set the admin password on first run:

   curl -s -X POST http://localhost:8080/api/v1/auth/setup \
     -H 'Content-Type: application/json' \
     -d '{"username":"admin","password":"<your-password>"}'

See https://ashmonger.github.io/skoed for full documentation.
